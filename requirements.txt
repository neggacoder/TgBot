# Зависимости телеграм-бота и веб-панели.
#   pip install -r requirements.txt
#
# Нижние границы версий проставлены везде: без них свежая мажорная версия
# (например, aiogram 4 или pydantic 1) молча ломает импорты на новой машине.
# Верхняя граница стоит только там, где мажорное обновление точно ломающее.

# --- ядро бота -------------------------------------------------------------
aiogram>=3.4,<4            # клиент Bot API; в 4.x ожидаются несовместимые изменения
aiomysql>=0.2.0            # асинхронный драйвер MySQL (db.py)
python-dotenv>=1.0.0       # чтение .env — им пользуются и бот, и панель

# --- картинки и голос ------------------------------------------------------
Pillow>=10.0               # стикеры-цитаты, карточки профиля, аватары
matplotlib>=3.7            # графики активности (activity_chart.py)
gTTS>=2.5                  # озвучка текста для команды «.скажи»
faster-whisper>=1.1,<2     # локальная транскрибация голосовых, аудио и видео
# gTTS 2.5 требует click<8.2, а huggingface-hub 1.x — click>=8.4. Эта
# совместимая ветка hub полностью покрывает faster-whisper и не ломает «.скажи».
huggingface-hub>=0.21,<1
click>=8.0,<8.2

# --- веб-панель (python -m webpanel) ---------------------------------------
fastapi>=0.110             # сам веб-фреймворк панели
uvicorn[standard]>=0.29    # ASGI-сервер; [standard] добавляет быстрый парсер http
pydantic>=2.0              # схемы запросов панели; FastAPI 0.110+ требует именно 2.x
argon2-cffi>=23.1          # argon2id-хеши паролей панели
itsdangerous>=2.1          # подписанные сессионные куки
python-multipart>=0.0.9    # приём форм и файлов: без него падают загрузка
                           # картинок и /api/send_photo. Напрямую нигде не
                           # импортируется — FastAPI дёргает его сам, поэтому
                           # пакет легко потерять при чистке зависимостей.

# Если панель на этой машине не нужна (только бот), блок выше можно не
# ставить — бот работает без него. Обратное неверно: панель ходит в ту же
# базу и использует aiogram для отправки сообщений.

# --- часовые пояса ----------------------------------------------------------
# Настройка «часовой пояс» (см. tz_settings.py) читает базу зон через
# zoneinfo. На Linux она обычно берётся из системного /usr/share/zoneinfo, но
# в slim-образах его нет, и ZoneInfo("Europe/Moscow") падает — бот тогда молча
# показывал бы всё в UTC. Пакет tzdata кладёт базу внутрь окружения.
tzdata>=2024.1

# --- тесты (python -m pytest tests/) ----------------------------------------
# На сервере не нужны: ставьте только если правите код. Тесты подменяют
# aiogram и aiomysql заглушками (tests/conftest.py), поэтому запускаются без
# .env, без MySQL и без токена бота.
pytest>=8.0
pyflakes>=3.2                # tests/test_no_undefined_names.py — ищет обращения
                           # к несуществующим именам и задвоенные объявления;
                           # без него этот тест пропускается
httpx>=0.27                # fastapi.testclient ходит через него
